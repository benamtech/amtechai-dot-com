{{- /* What every machine surface needs to know about one new page, from its content file: route, url, twin (its Markdown twin's path), okf (its bundle path, relative to /okf/), type (OKF), schema (WebPage or CollectionPage), h1 (as the page shows it), crumbs (the breadcrumb, home first), lastmod. */ -}} {{- $route := .Params.route | default (strings.TrimSuffix "/" .RelPermalink) -}} {{- $origin := site.BaseURL | strings.TrimSuffix "/" -}} {{- $url := printf "%s%s" $origin $route -}} {{- $h1 := .Params.h1 | default (printf "%s %s" .Params.h1a .Params.h1b) -}} {{- $type := "Web Page" }}{{ $schema := "WebPage" -}} {{- $crumbs := slice (dict "name" "Home" "url" (printf "%s/" $origin)) -}} {{- if .IsHome }}{{ $type = "Organization Overview" -}} {{- else if eq .Section "services" }}{{ $type = "Service Catalog" }}{{ $schema = "CollectionPage" -}} {{- else if and (eq .Section "industries") .IsSection }}{{ $type = "Industry" }}{{ $schema = "CollectionPage" -}} {{- else if eq .Section "industries" }}{{ $type = "Business Type" }}{{ $crumbs = $crumbs | append (dict "name" "Industries" "url" (printf "%s/industries" $origin)) -}} {{- else if eq .Section "departments" }}{{ $type = "Department" }}{{ end -}} {{- $name := .Params.label | default .Params.crumb | default (cond (eq .Section "services") "Services" (cond (and (eq .Section "industries") .IsSection) "Industries" ((index (split .Title ":") 0)))) -}} {{- if not .IsHome }}{{ $crumbs = $crumbs | append (dict "name" $name "url" $url) }}{{ end -}} {{- $twin := cond .IsHome "/index.md" (printf "%s.md" $route) -}} {{- $okf := cond .IsHome "/home.md" (printf "%s.md" $route) -}} {{- return dict "route" $route "url" (cond .IsHome (printf "%s/" $origin) $url) "twin" $twin "okf" $okf "type" $type "schema" $schema "h1" $h1 "name" $name "crumbs" $crumbs "lastmod" (partial "w/lastmod.html" .) -}}
